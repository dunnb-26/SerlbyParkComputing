    <script>
        function switchTab(yearId) {
            // 1. Hide all tab contents
            const contents = document.querySelectorAll('.tab-content');
            contents.forEach(content => {
                content.classList.remove('active');
            });

            // 2. Reset all nav buttons to inactive state
            const buttons = document.querySelectorAll('.nav-btn');
            buttons.forEach(btn => {
                btn.className = "nav-btn w-full flex items-center gap-3 px-4 py-3 text-left rounded-lg text-sm font-medium text-slate-600 hover:bg-slate-100 hover:text-[#232559] transition-colors";
                const iconBox = btn.querySelector('span');
                iconBox.className = "w-8 h-8 rounded bg-slate-200 text-slate-600 flex items-center justify-center font-bold transition-colors";
            });

            // 3. Show the target content
            document.getElementById(yearId).classList.add('active');

            // 4. Style the active button with Delta Serlby Park colors
            const activeBtn = document.getElementById(`btn-${yearId}`);
            activeBtn.className = "nav-btn w-full flex items-center gap-3 px-4 py-3 text-left rounded-lg text-sm font-bold text-[#232559] bg-[#E3A82B] transition-colors shadow-sm";
            
            const activeIconBox = activeBtn.querySelector('span');
            activeIconBox.className = "w-8 h-8 rounded bg-white/60 text-[#232559] flex items-center justify-center font-bold";
        }
    </script>
